#!name=YouTube 增强
#!desc=去除 YouTube 广告，支持后台播放、画中画
#!icon=https://iili.io/KQ67Yxa.png
#!author=gholts
#!homepage=https://github.com/gholts/surge/tree/main/modules
#!category=gholts
#!tag=去广告, YouTube
#!system=iOS

[Argument]
blockUpload = switch,true,tag=隐藏「上传」入口,desc=隐藏导航栏中的「上传」入口
blockShorts = switch,false,tag=隐藏「Shorts」入口,desc=隐藏导航栏中的「Shorts」入口
autoHd = switch,true,tag=自动最高画质,desc=优先使用最高可用画质，并阻止自动降低画质
blockGames = switch,true,tag=隐藏小游戏推荐,desc=隐藏 YouTube Playables（小游戏）推荐
blockVerticalLive = switch,false,tag=隐藏竖屏直播推荐,desc=隐藏各界面中的竖屏直播推荐
jumpAhead = switch,true,tag=启用 Jump Ahead,desc=启用原生「Jump Ahead」跳转按钮，并去除其推广提示
blockStore = switch,true,tag=隐藏商店和购物板块,desc=隐藏视频下方和频道主页中的创作者商店、购物板块
nativeDownload = switch,false,tag=原生下载（实验性）,desc=实验性原生下载功能；未设置画质时使用 720p，服务器限制可能仍然生效

[Script]
# 广告拦截：播放中广告插播请求
http-request ^https:\/\/youtubei\.googleapis\.com\/youtubei\/v1\/player\/ad_break(?:\?.*)?$ script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/request.js, tag=youtube.enhance.ad-break

# 原生下载（响应）
http-response ^https:\/\/youtubei\.googleapis\.com\/youtubei\/v1\/offline\/get_download_action(?:\?.*)?$ script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/response.js, requires-body=true, binary-body-mode=true, max-size=0, tag=youtube.enhance.download, argument="{"nativeDownload":{nativeDownload}}"

# 原生下载（播放请求）
http-request ^https:\/\/youtubei\.googleapis\.com\/youtubei\/v1\/player(?:\?.*)?$ script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/request.js, requires-body=true, binary-body-mode=true, max-size=0, tag=youtube.enhance.download-player, argument="{"nativeDownload":{nativeDownload}}"

# 主要响应处理：去广告、隐藏板块、后台播放、画中画等
http-response ^https:\/\/youtubei\.googleapis\.com\/youtubei\/v1\/(?:navigation\/resolve_url|browse|next|player|search|reel\/reel_watch_sequence|guide|account\/get_setting|get_watch|log_event|config)(?:\?.*)?$ script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/response.js, requires-body=true, binary-body-mode=true, max-size=0, tag=youtube.enhance.response, argument="{"blockUpload":{blockUpload},"blockShorts":{blockShorts},"blockGames":{blockGames},"blockVerticalLive":{blockVerticalLive},"jumpAhead":{jumpAhead},"autoHd":{autoHd},"blockStore":{blockStore}}"

# 视频流初始化（响应）
http-response ^https?:\/\/[\w-]+\.googlevideo\.com\/initplayback(?:\?.*)?$ script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/response.js, requires-body=true, binary-body-mode=true, max-size=0, tag=youtube.enhance.ump, argument="{"blockGames":{blockGames},"blockVerticalLive":{blockVerticalLive},"jumpAhead":{jumpAhead},"autoHd":{autoHd},"blockStore":{blockStore}}"

# 视频流初始化（请求）
http-request ^https?:\/\/[\w-]+\.googlevideo\.com\/initplayback(?:\?.*)?$ script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/request.js, requires-body=true, binary-body-mode=true, max-size=0, tag=youtube.enhance.init

# 自动最高画质：视频流请求
http-request ^https?:\/\/[\w-]+\.googlevideo\.com\/videoplayback script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/request.js, requires-body=true, binary-body-mode=true, max-size=0, tag=youtube.enhance.auto-hd, argument="{"autoHd":{autoHd}}"

# 日志上报请求处理
http-request ^https:\/\/youtubei\.googleapis\.com\/youtubei\/v1\/log_event script-path=https://raw.githubusercontent.com/gholts/surge/main/scripts/youtube/request.js, requires-body=true, binary-body-mode=true, max-size=0, tag=youtube.enhance.log-event

[Rewrite]
# 直接拒绝广告和统计请求
^https:\/\/www\.youtube\.com\/(pagead|pcs)\/ reject
^https:\/\/s\.youtube\.com\/api\/stats\/ads reject

[MITM]
hostname = *.googlevideo.com, youtubei.googleapis.com, www.youtube.com, s.youtube.com
